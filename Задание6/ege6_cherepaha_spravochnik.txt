                         ЧТО ТАКОЕ «ЧЕРЕПАХА»?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Представь себе черепаху, которая стоит на листе бумаги и держит в лапе
карандаш. Ты даёшь ей команды: «иди вперёд», «повернись», «подними карандаш».
Черепаха ходит и рисует линии.

В задании 6 тебе дают готовый код с такой черепахой, и нужно понять,
какую фигуру она нарисует, и найти её площадь или периметр.

Ты НЕ должен запускать программу. Ты должен «прокрутить» её в голове
(или на листочке) и понять, что получится.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
                    НАЧАЛЬНОЕ ПОЛОЖЕНИЕ ЧЕРЕПАХИ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Когда черепаха только создана (до каких-либо команд):

  • Стоит в точке (0, 0) — центр экрана
  • Смотрит ВПРАВО (на восток, угол 0°)
  • Карандаш ОПУЩЕН (рисует)

Это важно! Если тебе не сказали, куда повернули черепаху в начале,
значит она смотрит вправо.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
                    СТАНДАРТНЫЙ ШАБЛОН КОДА (как выглядит задание)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

from turtle import *

t = Turtle()       ← создаём черепаху, даём ей имя t
...команды...
done()             ← завершаем (иногда есть, иногда нет — не важно)

Все команды пишутся как t.команда(значение)
Например: t.fd(100)  или  t.left(90)


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
                         КОМАНДЫ ДВИЖЕНИЯ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

t.fd(N)      — иди ВПЕРЁД на N шагов (forward)
               N — это число (может быть любым: 50, 100, 200...)
               Черепаха идёт туда, КУДА СМОТРИТ.
               Пример: t.fd(100) → прошла 100 шагов вперёд

t.bk(N)      — иди НАЗАД на N шагов (backward)
               Черепаха идёт В ПРОТИВОПОЛОЖНУЮ сторону от того, куда смотрит.
               Пример: t.bk(50) → отошла на 50 шагов назад

t.forward(N) — то же самое, что t.fd(N), просто длинная запись
t.backward(N)— то же самое, что t.bk(N), просто длинная запись

ВАЖНО: если карандаш опущен (pen down) — черепаха рисует линию.
        если карандаш поднят (pen up) — черепаха идёт, но не рисует.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
                         КОМАНДЫ ПОВОРОТА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

t.left(угол)   — повернуться ВЛЕВО на указанный угол (против часовой стрелки)
t.lt(угол)     — то же самое, сокращение

t.right(угол)  — повернуться ВПРАВО на указанный угол (по часовой стрелке)
t.rt(угол)     — то же самое, сокращение

Угол измеряется в ГРАДУСАХ.

Примеры поворотов (черепаха изначально смотрит ВПРАВО →):

  t.left(90)  → теперь смотрит ВВЕРХ ↑
  t.left(180) → теперь смотрит ВЛЕВО ←
  t.left(270) → теперь смотрит ВНИЗ ↓
  t.left(360) → снова смотрит ВПРАВО → (полный круг)

  t.right(90)  → теперь смотрит ВНИЗ ↓
  t.right(180) → теперь смотрит ВЛЕВО ←
  t.right(270) → теперь смотрит ВВЕРХ ↑

ПОДСКАЗКА: повернуться влево на 90° = то же самое, что повернуться вправо на 270°.
           Повороты накапливаются! Два поворота влево на 45° = один поворот на 90°.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
               КАК ДВИГАЕТСЯ ЧЕРЕПАХА — СТОРОНЫ СВЕТА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Представь компас:

                        ВВЕРХ (север) ↑
                              │
             ВЛЕВО ← ─────────┼───────── → ВПРАВО
                              │
                        ВНИЗ (юг) ↓

Черепаха ВСЕГДА движется туда, куда «смотрит носом».

  Смотрит вправо →  + fd(N)  = идёт вправо,  X увеличивается на N
  Смотрит влево  ←  + fd(N)  = идёт влево,   X уменьшается на N
  Смотрит вверх  ↑  + fd(N)  = идёт вверх,   Y увеличивается на N
  Смотрит вниз   ↓  + fd(N)  = идёт вниз,    Y уменьшается на N

ПРИМЕР: Черепаха смотрит вправо. Команда t.fd(100).
        Она прошла 100 шагов вправо. Теперь её X-координата стала +100.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
                      КАРАНДАШ (pen) — РИСУЕТ ИЛИ НЕТ?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

t.up()       — ПОДНЯТЬ карандаш (pen up)
t.pu()       — то же самое, сокращение
t.penup()    — то же самое, длинная запись

  → Черепаха идёт, но линию НЕ рисует (как поднять ручку над бумагой)

t.down()     — ОПУСТИТЬ карандаш (pen down)
t.pd()       — то же самое, сокращение
t.pendown()  — то же самое, длинная запись

  → Черепаха идёт и рисует линию

По умолчанию карандаш ОПУЩЕН. Если в коде нет t.up() — значит черепаха
рисует всё время.

ПРИМЕР:
  t.fd(100)    ← рисует линию 100 шагов вправо
  t.up()       ← поднимаем карандаш
  t.fd(50)     ← идёт 50 шагов вправо, но НЕ рисует (пустое место)
  t.down()     ← опускаем карандаш
  t.fd(100)    ← снова рисует 100 шагов вправо


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
                      ПРОЧИЕ ПОЛЕЗНЫЕ КОМАНДЫ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

t.goto(x, y)      — ПРЫГНУТЬ в точку с координатами (x, y)
t.setpos(x, y)    — то же самое
  Черепаха мгновенно перемещается в нужную точку.
  Если карандаш опущен — нарисует прямую линию от текущей позиции до (x, y)!
  Если карандаш поднят — просто переместится без линии.

t.seth(угол)      — установить направление черепахи напрямую (setheading)
t.setheading(угол)— то же самое
  Углы: 0° = вправо, 90° = вверх, 180° = влево, 270° = вниз
  Пример: t.seth(90) → черепаха теперь смотрит ВВЕРХ

t.home()          — вернуться в начало (0, 0) и повернуться вправо
  Если карандаш опущен — нарисует линию от текущей точки до (0,0)!

t.circle(r)       — нарисовать окружность радиусом r
t.circle(r, угол) — нарисовать ДУГУ (часть окружности) на указанный угол
  Черепаха рисует окружность ПРОТИВ часовой стрелки.
  Центр окружности находится СЛЕВА от черепахи.
  Пример: t.circle(50) → полная окружность радиусом 50
           t.circle(50, 90) → дуга в 1/4 окружности


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
                         ЦИКЛЫ — КАК ЧИТАТЬ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

В задании часто встречаются циклы. Цикл — это «повторить N раз».

for i in range(N):   ← повторить N раз
    команда1         ← эти команды выполняются N раз (отступ = часть цикла!)
    команда2

Пример — нарисовать квадрат со стороной 100:

  for i in range(4):
      t.fd(100)
      t.left(90)

Это значит: 4 раза выполни «пройти 100 + повернуть на 90°»
Итог: квадрат! (4 стороны по 100, каждый раз поворот на 90°)

ВАЖНО: строки с одинаковым отступом (пробелы в начале) — внутри цикла.
        Если у строки нет отступа — она ВНЕ цикла.

Пример с вложенным циклом:

  for i in range(3):      ← внешний цикл, 3 раза
      for j in range(4):  ← внутренний цикл, 4 раза
          t.fd(50)
          t.left(90)
      t.left(120)

Читаем так: внешний цикл 3 раза. Каждый раз — рисуем квадрат (4×fd+left90),
потом поворачиваем на 120°.

                 КАК ЧИТАТЬ КОД — ПОШАГОВЫЙ МЕТОД
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Шаг 1. Найди начальное направление черепахи.
        Если не сказано — смотрит ВПРАВО.

Шаг 2. Иди по коду СВЕРХУ ВНИЗ, строка за строкой.

Шаг 3. Для каждой команды обновляй в голове (или на бумаге):
        - Куда смотрит черепаха?
        - Где она находится (примерно)?
        - Рисует или нет (карандаш поднят/опущен)?

Шаг 4. При встрече цикла — мысленно «разверни» его:
        напиши команды столько раз, сколько повторяется цикл.

Шаг 5. Нарисуй траекторию на листочке — это поможет понять фигуру.

                ТАБЛИЦА НАПРАВЛЕНИЙ — ШПАРГАЛКА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Текущее      │ Команда        │ Новое направление
направление  │                │
─────────────┼────────────────┼───────────────────
→ (вправо)   │ left(90)       │ ↑ (вверх)
→ (вправо)   │ left(180)      │ ← (влево)
→ (вправо)   │ left(270)      │ ↓ (вниз)
→ (вправо)   │ right(90)      │ ↓ (вниз)
→ (вправо)   │ right(180)     │ ← (влево)
→ (вправо)   │ right(270)     │ ↑ (вверх)
─────────────┼────────────────┼───────────────────
↑ (вверх)    │ left(90)       │ ← (влево)
↑ (вверх)    │ right(90)      │ → (вправо)
↑ (вверх)    │ left(180)      │ ↓ (вниз)
─────────────┼────────────────┼───────────────────
← (влево)    │ left(90)       │ ↓ (вниз)
← (влево)    │ right(90)      │ ↑ (вверх)
─────────────┼────────────────┼───────────────────
↓ (вниз)     │ left(90)       │ → (вправо)
↓ (вниз)     │ right(90)      │ ← (влево)

              ТИПИЧНЫЕ ФИГУРЫ В ЗАДАНИИ 6
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

КВАДРАТ (сторона A):
  for i in range(4):
      t.fd(A)
      t.left(90)    ← или right(90) — без разницы, оба дадут квадрат
  Площадь = A × A
  Периметр = 4 × A

ПРЯМОУГОЛЬНИК (стороны A и B):
  for i in range(2):
      t.fd(A)
      t.left(90)
      t.fd(B)
      t.left(90)
  Площадь = A × B
  Периметр = 2 × (A + B)

РАВНОСТОРОННИЙ ТРЕУГОЛЬНИК (сторона A):
  for i in range(3):
      t.fd(A)
      t.left(120)
  Площадь = (√3 / 4) × A²  ≈  0.433 × A²
  Периметр = 3 × A

ПРАВИЛЬНЫЙ ШЕСТИУГОЛЬНИК (сторона A):
  for i in range(6):
      t.fd(A)
      t.left(60)
  Площадь = (3√3 / 2) × A²  ≈  2.598 × A²
  Периметр = 6 × A

ПРАВИЛО: угол поворота при рисовании правильного N-угольника = 360° / N
  Треугольник: 360/3 = 120°
  Квадрат:     360/4 = 90°
  Шестиугольник: 360/6 = 60°
  и т.д.

         ПЕРЕМЕННЫЕ В КОДЕ — КАК СЧИТАТЬ ЗНАЧЕНИЯ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Иногда в коде есть переменные:

  a = 100
  b = a * 2
  t.fd(a)
  t.fd(b)

Читаем:
  a = 100           → a равно 100
  b = a * 2 = 200   → b равно 200
  t.fd(a) = t.fd(100) → идём 100 шагов
  t.fd(b) = t.fd(200) → идём 200 шагов

Арифметика в Python:
  *   — умножение     (2 * 3 = 6)
  /   — деление       (10 / 2 = 5)
  //  — целое деление (10 // 3 = 3, остаток отбрасывается)
  %   — остаток       (10 % 3 = 1)
  **  — степень       (2 ** 3 = 8)
  +   — сложение
  -   — вычитание

Переменная i в цикле for i in range(N) принимает значения 0, 1, 2, ..., N-1
Пример: range(4) → i будет 0, 1, 2, 3

                 RANGE — ВАЖНЫЕ ВАРИАНТЫ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

range(N)        → числа от 0 до N-1       (всего N чисел)
range(A, B)     → числа от A до B-1       (всего B-A чисел)
range(A, B, S)  → числа от A до B-1 с шагом S

Примеры:
  range(4)       → 0, 1, 2, 3         (4 значения)
  range(1, 5)    → 1, 2, 3, 4         (4 значения)
  range(0, 10, 2)→ 0, 2, 4, 6, 8     (5 значений, шаг 2)
  range(5, 0, -1)→ 5, 4, 3, 2, 1     (5 значений, идём назад)

              ЧАСТЫЕ ЛОВУШКИ И КАК ИХ ИЗБЕЖАТЬ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

ЛОВУШКА 1: Перепутать left и right
  → Нарисуй стрелочку на листочке и физически «поверни» её.
  → left = против часовой стрелки, right = по часовой стрелке.

ЛОВУШКА 2: Забыть, что карандаш опущен по умолчанию
  → Если нет t.up() в начале — всё рисуется с первого движения.

ЛОВУШКА 3: t.goto() рисует линию, если карандаш опущен!
  → Часто перед goto пишут t.up(), чтобы переместиться без линии.

ЛОВУШКА 4: bk(N) — черепаха идёт НАЗАД, но СМОТРИТ В ТУ ЖЕ СТОРОНУ
  → Направление черепахи НЕ меняется при bk()!
  → Черепаха смотрела вправо, сделала bk(100) — ушла влево, но смотрит вправо.

ЛОВУШКА 5: Переменная i в цикле меняется каждый раз
  → Если t.fd(i * 10), то в первый раз fd(0), потом fd(10), fd(20) и т.д.
  → Каждый шаг цикла — разная длина!

ЛОВУШКА 6: Цикл повторяет ВСЁ, что внутри с отступом
  → Проверяй отступы! Что без отступа — выполнится только ОДИН раз после цикла.

                  ПРИМЕР РАЗБОРА ЗАДАНИЯ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

ЗАДАНИЕ: Определить площадь фигуры, нарисованной программой:

  from turtle import *
  t = Turtle()
  for i in range(4):
      t.fd(100)
      t.left(90)

РАЗБОР:
  • Черепаха смотрит вправо.
  • Цикл повторяется 4 раза. Разворачиваем:

    Шаг 1: fd(100) → идём 100 вправо.   left(90) → смотрим вверх.
    Шаг 2: fd(100) → идём 100 вверх.   left(90) → смотрим влево.
    Шаг 3: fd(100) → идём 100 влево.   left(90) → смотрим вниз.
    Шаг 4: fd(100) → идём 100 вниз.    left(90) → смотрим вправо (вернулись).

  • Нарисован квадрат со стороной 100.
  • Площадь = 100 × 100 = 10 000

                  ФОРМУЛЫ ДЛЯ ЗАДАНИЯ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Квадрат (сторона a):
  Площадь   = a²
  Периметр  = 4a

Прямоугольник (стороны a, b):
  Площадь   = a × b
  Периметр  = 2(a + b)

Треугольник прямоугольный (катеты a, b):
  Площадь   = (a × b) / 2
  Гипотенуза = √(a² + b²)

Треугольник равносторонний (сторона a):
  Площадь   = (√3 / 4) × a²  ≈  0.433 × a²

Параллелограмм (основание a, высота h):
  Площадь   = a × h

Трапеция (основания a и b, высота h):
  Площадь   = ((a + b) / 2) × h

Окружность (радиус r):
  Длина     = 2 × π × r  ≈  6.283 × r
  Площадь   = π × r²     ≈  3.14159 × r²

                   БЫСТРАЯ ШПАРГАЛКА (всё самое важное)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

fd(N)       = вперёд N шагов
bk(N)       = назад N шагов
left(A)     = повернуть влево на A градусов
right(A)    = повернуть вправо на A градусов
up()        = поднять карандаш (не рисовать)
down()      = опустить карандаш (рисовать)
goto(x,y)   = прыгнуть в точку (рисует если карандаш опущен!)
seth(A)     = установить направление (0=вправо, 90=вверх, 180=влево, 270=вниз)
circle(r)   = окружность радиусом r

Начало: точка (0,0), смотрит вправо →, карандаш опущен.
left = против часовой.   right = по часовой.
bk не меняет направление взгляда!
range(N) → N повторений, i от 0 до N-1.
